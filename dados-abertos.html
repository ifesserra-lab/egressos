Como reproduzir

# requer os JSONs de mercado (Stack Overflow / IPCA) e a base de perfis (não pública)
python build_report.py            # gera todos os JSON + valida (QA + PII)
python build_report.py --publish  # copia a versão anonimizada p/ os sites
